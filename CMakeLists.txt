cmake_minimum_required(VERSION 3.16)
project(heveaReducedSphere_mma_static LANGUAGES CXX)

include(CTest)
include("${CMAKE_CURRENT_LIST_DIR}/cmake/WolframLibraryLink.cmake")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")

find_package(OpenMP)

add_executable(hevea_reduced_sphere hevea_reduced_sphere.cpp)
target_compile_options(hevea_reduced_sphere PRIVATE -O3)
if(OpenMP_CXX_FOUND)
  target_link_libraries(hevea_reduced_sphere PRIVATE OpenMP::OpenMP_CXX)
endif()
set_target_properties(hevea_reduced_sphere PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

add_library(heveaReducedSphere_mma_static SHARED hevea_reduced_sphere_wolfram.cpp)
target_include_directories(heveaReducedSphere_mma_static PRIVATE "${WOLFRAM_LIBRARYLINK_INCLUDE_DIR}")
target_link_libraries(heveaReducedSphere_mma_static PRIVATE dl)
set_target_properties(heveaReducedSphere_mma_static PROPERTIES
  OUTPUT_NAME "heveaReducedSphere_mma_static"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

  target_include_directories(numeric_kernel_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")

  target_compile_options(profile_tests PRIVATE -O2 -Wall -Wextra -Wpedantic)
  add_test(NAME profile_shortness_caps COMMAND profile_tests)
  add_executable(corrugation_tests tests/corrugation_tests.cpp)
  target_include_directories(corrugation_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_compile_options(corrugation_tests PRIVATE -O2 -Wall -Wextra -Wpedantic)
  add_test(NAME corrugation_phase_gluing_mesh COMMAND corrugation_tests)
